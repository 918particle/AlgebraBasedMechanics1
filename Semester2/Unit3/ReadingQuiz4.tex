\section{Memory Bank}

\begin{itemize}
\item $\vec{F} = I\vec{L} \times \vec{B}$ ... The Lorentz force on a current.
\item $\vec{F} = I \vec{L} \times \vec{B}$ ... Lorentz Force on a Current
\item $\vec{B} = (\mu_0 I)/(2\pi r)\hat{\phi}$ ... The magnetic field $B$ \textit{caused} by the current $I$ a distance $r$ away.  The field \textit{circles} the wire, so the direction of $\vec{B}$ is $\hat{\phi}$.
\item $\mu_0 = 4\pi \times 10^{-7}$ T m A$^{-1}$.
\item $\vec{B} = \mu_0 n I \hat{z}$ ... The B-field inside a solenoid.  $n=N/L$, the turns per unit length.
\item $\vec{B} = \mu_0 I/(2R) \hat{z}$ ... The B-field inside a circular loop with radius $R$.
\end{itemize}

\section{Force on an Element of Current}

\begin{enumerate}
\item A wire on a circuit board has a length L=4.7mm and carries a current I=2.23 $\mu$A in the positive y direction. A nearby circuit element generates a magnetic field in the vicinity of the wire given by 
\begin{equation}
\vec{B} = B_x\hat{i} + B_y\hat{j} + B_z\hat{k}
\end{equation}
with $B_x = 5$ G, $B_y = 4$ G, and $B_z = 3$ G.  (a) Calculate the force on the current element due to the field $\vec{B}$. (b) Compare the magnitude of this result to the maximum possible force on the current, $ILB$. \\ \vspace{5cm}
\end{enumerate}
