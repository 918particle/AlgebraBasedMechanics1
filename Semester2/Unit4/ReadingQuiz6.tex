\section{Memory Bank}

\begin{itemize}
\item $\tau = L/R$ ... The time-constant of an RL circuit.
\item $\tau = RC$ ... The time-constant of an RC circuit.
\item $v(t) = v_0 \sin(2\pi ft - \phi)$ ... Standard AC waveform, with amplitude $v_0$, in volts, frequency $f$, in Hertz, time $t$, in seconds, and phase $\phi$, in radians.
\item $2\pi f_0 = 1/\sqrt{LC}$ ... The resonance frequency, $f_0$, of an RLC circuit.
\item $V_{\rm rms} = I_{\rm rms} Z_{\rm tot}$ ... Ohm's Law for AC circuits, with total impedance $Z_{\rm tot}$.
\end{itemize}

\end{enumerate}
