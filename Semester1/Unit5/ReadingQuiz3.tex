\section{Memory Bank}

\begin{itemize}
\item $\vec{s} = -k\Delta\vec{x}$ ... Spring force
\item $\vec{f} = -\mu N\hat{i}$ ... Force of friction on a flat surface
\item $F_{\rm D} = \frac{1}{2}C\rho A v^2$ ... Drag force
\item $\Delta L/L_0 = (1/Y) F/A$ ... Relationship between
\begin{itemize}
\item $\Delta L$, change in length
\item $L_0$, original length
\item $F$, applied force
\item $A$, cross-sectional area
\item $Y$, Young's modulus
\end{itemize}
\end{itemize}

\section{Forces, II}

\begin{enumerate}
\item Suppose a spring has a k-value of $30$ N m$^{-1}$, and it is attached horizontally to a mass of 100 grams.  If the static coefficient of friction between the mass and the surface is 0.1, by how much can the spring be squeezed before the mass moves? \\ \vspace{3cm}
\item Consider Fig. \ref{fig:1} (left), in which a spring with k-value $k$ holds a mass $m$.  (a) Derive an expression for the displacement of the spring from its unstretched length, given $m$, $\theta$, and $k$. (b) Assume $m = 1.0$ kg, $\theta = 45$ degrees, and $k = 20$ N m$^{-1}$.  What is $\Delta x$, the displacement of the spring? (c) Now redo the exercise, but assume the surface has a coefficient of static friction of 0.33.  \\ \vspace{4cm}
\item (a) By what factor does the drag force on a car increase as it goes from 50 to 100 km hr$^{-1}$? (b) What is the drag force on a racecar with $C=0.2$, $m=700$ kg, in air with $\rho = 1.2$ kg m$^{-3}$, and area 3 m$^2$, at a speed of 150 km hr$^{-1}$? (c) What is the drag at 75 km hr$^{-1}$? \\ \vspace{4cm}
\item Consider Fig. \ref{fig:1} (right). TV broadcast antennas are the tallest artificial structures on Earth. In 1987, a 72.0-kg physicist placed himself and 400 kg of equipment at the top of one 610-m high antenna to perform gravity experiments. By how much was the antenna compressed, if we treat it like a steel cylinder 0.150 m in radius? For Young's modulus, use $210 \times 10^9$ N m$^{-2}$.
\end{enumerate}
