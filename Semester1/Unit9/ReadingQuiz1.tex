\section{Memory Bank}

\begin{itemize}
\item $\hat{i} \cdot \hat{i} = 1$, $\hat{i} \cdot \hat{j} = 0$ ... The dot product, unit vectors
\item $\vec{u} \cdot \vec{v} = uv\cos\theta$ ... The dot product of two vectors is the product of the magnitudes times the cosine of the angle between them.
\item $\hat{i} \times \hat{j} = \hat{k}$, $\hat{k} \times \hat{i} = \hat{j}$, $\hat{j} \times \hat{k} = \hat{i}$ ... The \textit{cross-product}, unit vectors
\item $\hat{j} \times \hat{i} = -\hat{k}$, $\hat{i} \times \hat{k} = -\hat{j}$, $\hat{k} \times \hat{j} = -\hat{i}$ ... When the unit vectors are not in alphabetical order, there is a minus sign.
\item $\tau = \vec{r} \times \vec{F}$ ... Definition of torque.
\item $|\tau| = rF\sin\theta$ ... The magnitude of the torque is the product of $r$ and $F$, times the sine of the angle between them.  The distance $r$ is between the force and the pivot point.
\item Torque is the angular version of force.  It is the result of a force $F$, separated from a pivot by a distance $r$, that rotates the system about the pivot.  The angle between the force $F$ and the distance $r$ is $\theta$.
\item \textbf{For sytems in static equilibrium, (1) the net force is zero, and (2) the net torque is zero.}
\end{itemize}

\section{Dot Product, and Cross Product}

\begin{enumerate}
\item Calculate the dot product:
\begin{itemize}
\item $\vec{a} = 2\hat{i} - 2\hat{j}$, $\vec{b} = -4\hat{i} + 4\hat{j}$.
\item $\vec{a} = 2\hat{i}$, $\vec{b} = -4\hat{k}$.
\end{itemize}
\item Calculate the cross product:
\begin{itemize}
\item $\vec{a} = 2\hat{i}$, $\vec{b} = -4\hat{k}$.
\item $\vec{a} = 2\hat{i} + 2\hat{j}$, $\vec{b} = -2\hat{i} - 2\hat{j}$.
\end{itemize}
\end{enumerate}
